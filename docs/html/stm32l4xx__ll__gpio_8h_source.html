781__STATIC_INLINE void LL_GPIO_LockPin(GPIO_TypeDef *GPIOx, uint32_t PinMask)
782{
783 __IO uint32_t temp;
784 WRITE_REG(GPIOx->LCKR, GPIO_LCKR_LCKK | PinMask);
785 WRITE_REG(GPIOx->LCKR, PinMask);
786 WRITE_REG(GPIOx->LCKR, GPIO_LCKR_LCKK | PinMask);
787 /* Read LCKK register. This read is mandatory to complete key lock sequence */
788 temp = READ_REG(GPIOx->LCKR);
789 (void) temp;
790}
